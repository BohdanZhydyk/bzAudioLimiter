cmake_minimum_required(VERSION 3.16)
project(bzAudioLimiterDSP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(bzAudioLimiterVST SHARED bz_vst.cpp)
set_target_properties(bzAudioLimiterVST PROPERTIES PREFIX "" OUTPUT_NAME "bzAudioLimiterVST-0.2.2")
if(WIN32)
  target_link_libraries(bzAudioLimiterVST PRIVATE kernel32 advapi32)
endif()
add_executable(test_vst ../tests/test_vst.cpp bz_vst.cpp)
add_executable(test_extreme ../tests/test_extreme.cpp bz_vst.cpp)
enable_testing()
add_test(NAME vst_realtime_callback COMMAND test_vst)
add_test(NAME realtime_boundaries COMMAND test_extreme)
